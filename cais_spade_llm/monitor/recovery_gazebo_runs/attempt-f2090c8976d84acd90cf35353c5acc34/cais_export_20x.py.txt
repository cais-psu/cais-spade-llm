from __future__ import annotations

import hashlib
import json
from pathlib import Path
import shutil
import subprocess
import time

out = Path(Path('/tmp/cais_acceptance_output_path').read_text().strip())
source = out/'assembly.mp4'
temporary = out/'assembly-20x.partial.mp4'
target = out/'assembly-20x.mp4'
validation = out/'assembly-20x-validation.json'
assert not target.exists() and not temporary.exists()
accepted = json.loads((out/'success.json').read_text())
assert accepted['accepted'] and accepted['validated']
original = json.loads((out/'video_validation.json').read_text())
assert original['validated']
source_hash = hashlib.sha256(source.read_bytes()).hexdigest()
filter_text = (r'select=not(mod(n\,20)),setpts=N/(15*TB),'
               'drawbox=x=8:y=8:w=405:h=42:color=black:t=fill,'
               "drawtext=fontfile=/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf:"
               "text='20x speed | Gazebo assembly':x=16:y=18:fontsize=20:fontcolor=white")
command = ['ffmpeg','-hide_banner','-loglevel','error','-nostdin','-y',
           '-threads','2','-filter_threads','1','-i',str(source),'-an',
           '-vf',filter_text,'-r','15','-c:v','h264_nvenc','-preset','p1',
           '-rc','vbr','-cq','23','-b:v','0','-pix_fmt','yuv420p',
           '-bf','0','-delay','0','-zerolatency','1','-movflags','+faststart',
           '-progress',str(out/'assembly-20x.encoder.progress'),str(temporary)]
try:
    with (out/'assembly-20x.encoder.log').open('w') as log:
        subprocess.run(command,stdout=subprocess.DEVNULL,stderr=log,check=True,timeout=600)
    decoded = subprocess.run(['ffmpeg','-hide_banner','-loglevel','error','-xerror',
        '-nostdin','-threads','2','-i',str(temporary),'-map','0:v:0',
        '-progress','pipe:1','-nostats','-f','null','-'],
        capture_output=True,text=True,check=True,timeout=120)
    (out/'assembly-20x.decode.log').write_text(decoded.stdout+decoded.stderr)
    counts = [int(line.split('=',1)[1]) for line in decoded.stdout.splitlines() if line.startswith('frame=')]
    expected_frames = (original['full_frames_decoded'] - 1)//20 + 1
    assert counts and counts[-1] == expected_frames,(counts,expected_frames)
    probe = json.loads(subprocess.check_output(['ffprobe','-v','error','-select_streams','v:0',
        '-show_entries','stream=codec_name,width,height,avg_frame_rate,nb_frames:format=duration',
        '-of','json',str(temporary)],text=True))
    stream = probe['streams'][0];duration = float(probe['format']['duration'])
    assert [stream['width'],stream['height']] == original['size']
    assert stream['avg_frame_rate'] == '15/1'
    assert abs(duration-original['duration_sec']/20) <= 1/15+0.001
    assert hashlib.sha256(source.read_bytes()).hexdigest() == source_hash
    result = {'validated':True,'run_id':accepted['run_id'],'video':target.name,
              'source_video':source.name,'source_sha256':source_hash,
              'speed':20,'fps':15,'size':original['size'],'duration_sec':duration,
              'source_duration_sec':original['duration_sec'],'decoded_frames':counts[-1],
              'source_frames':original['full_frames_decoded'],'encoder':'h264_nvenc',
              'label':'20x speed | Gazebo assembly','created_at_unix':time.time(),
              'command':command,'video_sha256':hashlib.sha256(temporary.read_bytes()).hexdigest()}
    temporary.replace(target)
    validation.write_text(json.dumps(result,indent=2))
    accepted.setdefault('additional_videos',{})['20x'] = {
        'video':target.name,'speed':20,'duration_sec':duration,'validated':True,
        'validation':validation.name}
    (out/'success.json').write_text(json.dumps(accepted,indent=2))
    shutil.copyfile(Path(__file__),out/'cais_export_20x.py.txt')
    print(json.dumps({k:v for k,v in result.items() if k not in ['command']},indent=2),flush=True)
finally:
    temporary.unlink(missing_ok=True)
