from __future__ import annotations

import json
import math

root = Path.cwd()
out = Path(Path('/tmp/cais_acceptance_output_path').read_text().strip())
run = json.loads((out/'run.json').read_text())

comparison = json.loads((out/'speed_comparison.json').read_text())
previous, current = comparison['previous'], comparison['current']
base = []
for transition in run['transitions']:
    ack = transition['acknowledgement']
    if ack['resource_id'] != 'KMR':
        continue
    for operation in transition.get('observations', {}).get('observations', {}).get('operations', []):
        if operation.get('operation') != '/KMR/move_base':
            continue
        samples = operation.get('base_motion_samples', [])
        held = [row for row in samples if row.get('clearance_wait_reason')]
        # Timeout records retain the proposed vector; the executor publishes zero and returns.
        moving = [row for row in samples if not row.get('clearance_wait_reason')
                  and any(abs(v) > 1e-9 for v in row.get('command_velocity', []))]
        failures = [row for row in moving if row.get('map_available') is not True
                    or row.get('collision_checks_bypassed') is not False
                    or row.get('transport_clearance', {}).get('checked_states', 0) < 1
                    or row.get('transport_clearance', {}).get('collision_checks_bypassed') is not False]
        base.append({'task_id': ack['task_id'], 'function': ack['event_name'], 'part': ack.get('part_name'),
                     'destination': ack['parameters'].get('target_resource'),
                     'samples': len(samples), 'moving_samples': len(moving),
                     'clearance_hold_samples': len(held),
                     'clearance_hold_reasons': sorted({row['clearance_wait_reason'] for row in held}),
                     'moving_samples_without_clearance': len(failures),
                     'maximum_odometry_wall_age_sec': max((row.get('odometry_age_sec', 0) for row in samples), default=0),
                     'maximum_odometry_sim_age_sec': max((row.get('odometry_sim_age_sec', 0) for row in samples), default=0),
                     'final_feedback': operation.get('feedback'), 'success': operation.get('success')})
clearance = {'base_actions': base, 'moving_samples': sum(row['moving_samples'] for row in base),
             'moving_samples_without_clearance': sum(row['moving_samples_without_clearance'] for row in base),
             'clearance_hold_samples': sum(row['clearance_hold_samples'] for row in base),
             'held_sample_interpretation': 'clearance_wait_reason identifies a timeout branch that calls _stop() and returns before cmd_pub.publish(gated). command_velocity in these records is the proposed vector, not dispatched output. Existing test_kmr_stop_does_not_refresh_commands_or_resume_cancellation coverage verifies zero publication on timeout.',
             'contact_counts': acceptance['contact_counts'],
             'KMR_cartesian': acceptance['KMR_cartesian'],
             'scope': 'Recorded map, stopping-motion, parked-arm, and payload validity evidence. No numerical minimum surface distance was reported by GetStateValidity.'}
(out/'clearance_audit.json').write_text(json.dumps(clearance, indent=2))
assert acceptance['accepted']
assert len(base) == 22 and all(row['success'] for row in base)
assert clearance['moving_samples_without_clearance'] == 0
assert acceptance['contact_counts']['ur_equipment'] == 0
assert acceptance['contact_counts']['ur_robot'] == 0

    seconds = round(value)
    return f'{seconds//60}m{seconds%60:02d}s'
